{{- /* Link render hook. External links open in a new tab with rel="noopener" and carry a small icon, so a resident knows before clicking that they are leaving the association's site — which matters most on the pages that link to the county, the town, and the payment portal. Internal links pass through untouched. */ -}} {{- $u := urls.Parse .Destination -}} {{- $external := and $u.Host (ne $u.Host (urls.Parse site.BaseURL).Host) -}} {{- $isMailto := hasPrefix .Destination "mailto:" -}} {{- $isTel := hasPrefix .Destination "tel:" -}} {{- if and $external (not $isMailto) (not $isTel) -}} {{ .Text | safeHTML -}} {{ partial "ui/icon.html" "external" }} (opens in a new tab) {{- else -}} {{ .Text | safeHTML }} {{- end -}}
